void insertionsort(int a[],int n)
{
    
int i,j,temp;
for(i=1;i<n;i++)
{
      temp=a[i];
    for(j=i-1;j>=0 && temp<a[j];j--)
    {
            a[j+1]=a[j];
    }
a[j+1]=temp;
printf("\n PASS - %d :",i);
display(a,n);
}
}
void display(int a[],int n)
{
int k;
for(k=0;k<n;k++)
printf("%d\t",a[k]);
}
