void selectionsort(int a[],int n)
{
int i,j,t,min;
for(i=0;i<n-1;i++)
{
 min=i;
for(j=i+1;j<n;j++)
{
 if(a[j]<a[min])
 {
  min=j;
 }
}
if(min!=i)   // small is its place
{
t=a[i];
a[i]=a[min];
a[min]=t;
}
printf("\n PASS - %d :",i+1);
display(a,n);
}
}
void display(int a[],int n)
{
int k;
for(k=0;k<n;k++)
printf("%d\t",a[k]);
}
